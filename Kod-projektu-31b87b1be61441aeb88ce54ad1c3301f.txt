using System;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.Text.RegularExpressions;
using System.Threading.Tasks;

namespace VideoPlayer
{
    /// <summary>Wspólne narzędzia FFmpeg używane przez eksport i operacje na nagraniu.</summary>
    internal sealed class FfmpegService
    {
        public string ExecutablePath { get; }

        public FfmpegService(string? baseDirectory = null)
        {
            ExecutablePath = Path.Combine(baseDirectory ?? AppContext.BaseDirectory, "ffmpeg", "ffmpeg.exe");
        }

        public bool IsAvailable => File.Exists(ExecutablePath);

        public ProcessStartInfo CreateStartInfo(bool redirectStandardOutput = false, bool redirectStandardError = true)
        {
            return new ProcessStartInfo
            {
                FileName = ExecutablePath,
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardOutput = redirectStandardOutput,
                RedirectStandardError = redirectStandardError
            };
        }

        public async Task<(double Duration, bool HasAudio)> ProbeAsync(string sourcePath)
        {
            if (!IsAvailable)
                throw new FileNotFoundException("Nie znaleziono ffmpeg.exe.", ExecutablePath);

            var psi = CreateStartInfo();
            psi.ArgumentList.Add("-hide_banner");
            psi.ArgumentList.Add("-i");
            psi.ArgumentList.Add(sourcePath);

            using var process = new Process { StartInfo = psi };
            process.Start();
            string output = await process.StandardError.ReadToEndAsync().ConfigureAwait(false);
            await process.WaitForExitAsync().ConfigureAwait(false);

            double duration = MediaInfoService.ParseDuration(output);
            if (duration <= 0)
                throw new InvalidOperationException("FFmpeg nie rozpoznał czasu trwania pliku. Sprawdź, czy nagranie jest poprawne i obsługiwane.");

            bool hasAudio = Regex.IsMatch(output, @"Stream #\S+.*?: Audio:");
            return (duration, hasAudio);
        }
    }
}

using Microsoft.Data.Sqlite;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;

namespace VideoPlayer
{
    /// <summary>
    /// Stores application history and timeline markers in a portable SQLite database.
    /// The database is placed beside the application executable (or in the supplied directory).
    /// </summary>
    internal sealed class HistoryService
    {
        private readonly string _directory;
        private readonly string _databasePath;
        private readonly string _connectionString;

        public HistoryService(string? directory = null)
        {
            _directory = directory ?? AppContext.BaseDirectory;
            Directory.CreateDirectory(_directory);
            _databasePath = Path.Combine(_directory, "VideoPlayer.db");
            _connectionString = new SqliteConnectionStringBuilder
            {
                DataSource = _databasePath,
                Mode = SqliteOpenMode.ReadWriteCreate,
                Cache = SqliteCacheMode.Shared
            }.ToString();

            bool databaseAlreadyExisted = File.Exists(_databasePath);
            InitializeDatabase();
            if (!databaseAlreadyExisted)
                ImportLegacyJson();
        }

        public AppHistoryData Load()
        {
            var data = new AppHistoryData();
            using var connection = OpenConnection();
            using var command = connection.CreateCommand();
            command.CommandText = "SELECT Kind, Path FROM RecentItems ORDER BY Kind, Position;";
            using var reader = command.ExecuteReader();
            while (reader.Read())
            {
                string kind = reader.GetString(0);
                string path = reader.GetString(1);
                if (!File.Exists(path)) continue;
                if (kind == "file" && data.RecentFiles.Count < 10 && !data.RecentFiles.Contains(path, StringComparer.OrdinalIgnoreCase))
                    data.RecentFiles.Add(path);
                else if (kind == "screenshot" && data.RecentScreenshots.Count < 30 && !data.RecentScreenshots.Contains(path, StringComparer.OrdinalIgnoreCase))
                    data.RecentScreenshots.Add(path);
            }
            return data;
        }

        public void Save(AppHistoryData data)
        {
            using var connection = OpenConnection();
            using var transaction = connection.BeginTransaction();
            using (var delete = connection.CreateCommand())
            {
                delete.Transaction = transaction;
                delete.CommandText = "DELETE FROM RecentItems;";
                delete.ExecuteNonQuery();
            }

            InsertRecentItems(connection, transaction, "file", data.RecentFiles.Where(File.Exists).Distinct(StringComparer.OrdinalIgnoreCase).Take(10));
            InsertRecentItems(connection, transaction, "screenshot", data.RecentScreenshots.Where(File.Exists).Distinct(StringComparer.OrdinalIgnoreCase).Take(30));
            transaction.Commit();
        }

        public List<TimelineMarker> LoadMarkers(string sourcePath)
        {
            var markers = new List<TimelineMarker>();
            string key = NormalizePath(sourcePath);
            using var connection = OpenConnection();
            using var command = connection.CreateCommand();
            command.CommandText = @"SELECT Id, Seconds, Description FROM TimelineMarkers
                                    WHERE SourcePath = $sourcePath ORDER BY Seconds;";
            command.Parameters.AddWithValue("$sourcePath", key);
            using var reader = command.ExecuteReader();
            while (reader.Read())
            {
                if (!Guid.TryParse(reader.GetString(0), out Guid id)) id = Guid.NewGuid();
                markers.Add(new TimelineMarker
                {
                    Id = id,
                    Seconds = reader.GetDouble(1),
                    Description = reader.GetString(2)
                });
            }
            return markers;
        }

        public void SaveMarkers(string sourcePath, IEnumerable<TimelineMarker> markers)
        {
            string key = NormalizePath(sourcePath);
            using var connection = OpenConnection();
            using var transaction = connection.BeginTransaction();
            using (var delete = connection.CreateCommand())
            {
                delete.Transaction = transaction;
                delete.CommandText = "DELETE FROM TimelineMarkers WHERE SourcePath = $sourcePath;";
                delete.Parameters.AddWithValue("$sourcePath", key);
                delete.ExecuteNonQuery();
            }

            foreach (TimelineMarker marker in markers
                .Where(m => double.IsFinite(m.Seconds) && m.Seconds >= 0)
                .OrderBy(m => m.Seconds))
            {
                using var insert = connection.CreateCommand();
                insert.Transaction = transaction;
                insert.CommandText = @"INSERT INTO TimelineMarkers (Id, SourcePath, Seconds, Description)
                                       VALUES ($id, $sourcePath, $seconds, $description);";
                insert.Parameters.AddWithValue("$id", marker.Id.ToString("D"));
                insert.Parameters.AddWithValue("$sourcePath", key);
                insert.Parameters.AddWithValue("$seconds", marker.Seconds);
                insert.Parameters.AddWithValue("$description", marker.Description ?? string.Empty);
                insert.ExecuteNonQuery();
            }
            transaction.Commit();
        }

        private SqliteConnection OpenConnection()
        {
            var connection = new SqliteConnection(_connectionString);
            connection.Open();
            return connection;
        }

        private void InitializeDatabase()
        {
            using var connection = OpenConnection();
            using var command = connection.CreateCommand();
            command.CommandText = @"
                PRAGMA journal_mode = WAL;
                CREATE TABLE IF NOT EXISTS RecentItems (
                    Kind TEXT NOT NULL,
                    Path TEXT NOT NULL,
                    Position INTEGER NOT NULL,
                    PRIMARY KEY (Kind, Path)
                );
                CREATE INDEX IF NOT EXISTS IX_RecentItems_KindPosition ON RecentItems (Kind, Position);
                CREATE TABLE IF NOT EXISTS TimelineMarkers (
                    Id TEXT NOT NULL,
                    SourcePath TEXT NOT NULL,
                    Seconds REAL NOT NULL,
                    Description TEXT NOT NULL DEFAULT '',
                    PRIMARY KEY (SourcePath, Id)
                );
                CREATE INDEX IF NOT EXISTS IX_TimelineMarkers_SourceSeconds ON TimelineMarkers (SourcePath, Seconds);
            ";
            command.ExecuteNonQuery();
        }

        private void InsertRecentItems(SqliteConnection connection, SqliteTransaction transaction, string kind, IEnumerable<string> paths)
        {
            int position = 0;
            foreach (string path in paths)
            {
                using var command = connection.CreateCommand();
                command.Transaction = transaction;
                command.CommandText = "INSERT OR REPLACE INTO RecentItems (Kind, Path, Position) VALUES ($kind, $path, $position);";
                command.Parameters.AddWithValue("$kind", kind);
                command.Parameters.AddWithValue("$path", path);
                command.Parameters.AddWithValue("$position", position++);
                command.ExecuteNonQuery();
            }
        }

        private void ImportLegacyJson()
        {
            // One-time import of the previous AppData JSON files, so an update does not lose history.
            string legacyDirectory = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "VideoPlayer");
            string historyPath = Path.Combine(legacyDirectory, "history.json");
            string markersPath = Path.Combine(legacyDirectory, "timeline-markers.json");

            if (File.Exists(historyPath))
            {
                try
                {
                    var history = JsonSerializer.Deserialize<AppHistoryData>(File.ReadAllText(historyPath));
                    if (history != null) Save(history);
                }
                catch { /* Preserve the database and continue if an old JSON file is invalid. */ }
            }

            if (File.Exists(markersPath))
            {
                try
                {
                    var db = JsonSerializer.Deserialize<MarkerDatabase>(File.ReadAllText(markersPath));
                    if (db != null)
                    {
                        foreach (var pair in db.Files)
                        {
                            using var connection = OpenConnection();
                            using var transaction = connection.BeginTransaction();
                            foreach (TimelineMarker marker in pair.Value ?? new List<TimelineMarker>())
                            {
                                if (!double.IsFinite(marker.Seconds) || marker.Seconds < 0) continue;
                                using var command = connection.CreateCommand();
                                command.Transaction = transaction;
                                command.CommandText = @"INSERT OR REPLACE INTO TimelineMarkers (Id, SourcePath, Seconds, Description)
                                                        VALUES ($id, $sourcePath, $seconds, $description);";
                                command.Parameters.AddWithValue("$id", marker.Id.ToString("D"));
                                command.Parameters.AddWithValue("$sourcePath", NormalizePath(pair.Key));
                                command.Parameters.AddWithValue("$seconds", marker.Seconds);
                                command.Parameters.AddWithValue("$description", marker.Description ?? string.Empty);
                                command.ExecuteNonQuery();
                            }
                            transaction.Commit();
                        }
                    }
                }
                catch { /* Do not prevent startup because an old marker file cannot be imported. */ }
            }
        }

        private static string NormalizePath(string path) => Path.GetFullPath(path)
            .TrimEnd(Path.DirectorySeparatorChar).ToUpperInvariant();
    }

    internal sealed class AppHistoryData
    {
        public List<string> RecentFiles { get; set; } = new();
        public List<string> RecentScreenshots { get; set; } = new();
    }

    internal sealed class TimelineMarker
    {
        public Guid Id { get; set; } = Guid.NewGuid();
        public double Seconds { get; set; }
        public string Description { get; set; } = "";
        public override string ToString()
        {
            TimeSpan time = TimeSpan.FromSeconds(Math.Max(0, Seconds));
            return $"{(int)time.TotalHours:00}:{time.Minutes:00}:{time.Seconds:00} — {Description}";
        }
    }

    internal sealed class MarkerDatabase
    {
        public Dictionary<string, List<TimelineMarker>> Files { get; set; } = new(StringComparer.OrdinalIgnoreCase);
    }
}

using System;
using System.ComponentModel;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.IO.Pipes;
using System.Linq;
using System.Text;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Controls.Primitives;

using Forms = System.Windows.Forms;

namespace VideoPlayer
{
    public partial class MainWindow
    {
        private async void ExportClip_Click(
            object sender,
            RoutedEventArgs e)
        {
            if (_exporting)
                return;

            if (string.IsNullOrWhiteSpace(_sourceFilePath) ||
                !File.Exists(_sourceFilePath))
            {
                MessageBox.Show(
                    this,
                    "Najpierw otwórz lokalny plik filmu.",
                    "Eksport fragmentu");
                return;
            }

            if (!_clipStart.HasValue ||
                !_clipEnd.HasValue ||
                _clipEnd.Value <= _clipStart.Value)
            {
                MessageBox.Show(
                    this,
                    "Zaznacz początek i koniec fragmentu.",
                    "Eksport fragmentu");
                return;
            }

            string ffmpegPath = _ffmpegService.ExecutablePath;

            if (!File.Exists(ffmpegPath))
            {
                MessageBox.Show(
                    this,
                    "Nie znaleziono ffmpeg.exe w folderze ffmpeg.",
                    "Brak FFmpeg",
                    MessageBoxButton.OK,
                    MessageBoxImage.Warning);
                return;
            }

            // Zapamiętujemy źródło i zakres na czas eksportu.
            string sourcePath = _sourceFilePath;
            double start = _clipStart.Value;
            double duration = _clipEnd.Value - start;
            bool streamCopyRequested = StreamCopyCheckBox.IsChecked == true;

            if (streamCopyRequested)
            {
                MessageBoxResult confirm = MessageBox.Show(
                    this,
                    "Tryb bez ponownego kodowania jest bardzo szybki, ale początek i koniec mogą zostać dopasowane do klatek kluczowych. Niektóre kodeki mogą nie działać poprawnie w MP4. Kontynuować?",
                    "Eksport bez ponownego kodowania",
                    MessageBoxButton.YesNo,
                    MessageBoxImage.Warning);
                if (confirm != MessageBoxResult.Yes)
                    return;
            }

            var dialog = new Microsoft.Win32.SaveFileDialog
            {
                Title = "Zapisz zaznaczony fragment",
                Filter = "Film MP4 (*.mp4)|*.mp4",
                DefaultExt = ".mp4",
                AddExtension = true,
                OverwritePrompt = true,
                FileName =
                    Path.GetFileNameWithoutExtension(sourcePath) +
                    "_fragment.mp4"
            };

            if (dialog.ShowDialog(this) != true)
                return;

            string outputPath = Path.GetFullPath(dialog.FileName);

            if (string.Equals(
                Path.GetFullPath(sourcePath),
                outputPath,
                StringComparison.OrdinalIgnoreCase))
            {
                MessageBox.Show(
                    this,
                    "Wybierz inną nazwę niż nazwa filmu źródłowego.",
                    "Eksport fragmentu",
                    MessageBoxButton.OK,
                    MessageBoxImage.Warning);
                return;
            }

            string temporaryPath = Path.Combine(
                Path.GetDirectoryName(outputPath)!,
                "VideoPlayer_export_" +
                Guid.NewGuid().ToString("N") +
                ".mp4");

            _exporting = true;
            _cancelExportRequested = false;
            _exportStopwatch = Stopwatch.StartNew();
            _activeOperationName = Path.GetFileName(sourcePath);
            _activeOperationDuration = duration;

            ExportClipButton.IsEnabled = false;
            RemoveClipButton.IsEnabled = false;
            ExportProfileBox.IsEnabled = false;
            StreamCopyCheckBox.IsEnabled = false;
            CancelExportButton.Visibility = Visibility.Visible;
            CancelExportButton.IsEnabled = true;
            ExportProgressBar.Value = 0;
            ExportProgressText.Text = "Analiza źródła i przygotowanie eksportu…";
            ExportOperationText.Text = $"Plik: {_activeOperationName} | Zakres: {FormatTime(start)}–{FormatTime(start + duration)} | Długość: {FormatTime(duration)}";

            Process? process = null;

            try
            {
                var startInfo = _ffmpegService.CreateStartInfo(
                    redirectStandardOutput: true,
                    redirectStandardError: true);

                bool streamCopy = StreamCopyCheckBox.IsChecked == true;
                string profile = (ExportProfileBox.SelectedItem as ComboBoxItem)?.Tag?.ToString() ?? "standard";

                var arguments = new System.Collections.Generic.List<string>
                {
                    "-hide_banner", "-loglevel", "error", "-nostdin", "-y",
                    "-progress", "pipe:1", "-stats_period", "0.25", "-nostats",
                    "-ss", start.ToString("R", CultureInfo.InvariantCulture),
                    "-i", sourcePath,
                    "-t", duration.ToString("R", CultureInfo.InvariantCulture),
                    "-map", "0:v:0", "-map", "0:a:0?"
                };

                if (streamCopy)
                {
                    // Tryb bez ponownego kodowania: szybki, ale cięcie jest ograniczone
                    // do granic pakietów/klatek kluczowych, a kodeki muszą pasować do MP4.
                    arguments.AddRange(new[] { "-c:v", "copy", "-c:a", "copy", "-avoid_negative_ts", "make_zero" });
                }
                else
                {
                    string preset;
                    string crf;
                    switch (profile)
                    {
                        case "fast": preset = "veryfast"; crf = "22"; break;
                        case "small": preset = "slow"; crf = "27"; break;
                        case "quality": preset = "slow"; crf = "16"; break;
                        default: preset = "medium"; crf = "20"; break;
                    }

                    arguments.AddRange(new[]
                    {
                        "-c:v", "libx264", "-preset", preset, "-crf", crf,
                        "-vf", "pad=ceil(iw/2)*2:ceil(ih/2)*2", "-pix_fmt", "yuv420p",
                        "-c:a", "aac", "-b:a", "192k"
                    });
                }

                arguments.AddRange(new[] { "-sn", "-dn", "-map_chapters", "-1", "-movflags", "+faststart", temporaryPath });

                foreach (string argument in arguments)
                    startInfo.ArgumentList.Add(argument);

                process = new Process
                {
                    StartInfo = startInfo,
                    EnableRaisingEvents = true
                };

                if (!process.Start())
                    throw new InvalidOperationException(
                        "Nie udało się uruchomić FFmpeg.");

                _exportProcess = process;

                // Oba strumienie czytamy równolegle, aby uniknąć
                // zablokowania procesu przez pełny bufor.
                Task progressTask =
                    ReadFfmpegProgressAsync(process, duration);

                Task<string> errorTask =
                    process.StandardError.ReadToEndAsync();

                await process.WaitForExitAsync();
                await progressTask;

                string errors = await errorTask;

                if (_cancelExportRequested)
                    throw new OperationCanceledException(
                        "Eksport został anulowany.");

                if (process.ExitCode != 0)
                {
                    if (errors.Length > 4000)
                        errors = errors.Substring(errors.Length - 4000);

                    throw new InvalidOperationException(
                        string.IsNullOrWhiteSpace(errors)
                            ? "FFmpeg zakończył eksport błędem."
                            : errors.Trim());
                }

                if (!File.Exists(temporaryPath) ||
                    new FileInfo(temporaryPath).Length == 0)
                {
                    throw new IOException(
                        "Nie powstał poprawny plik wynikowy.");
                }

                // Dopiero teraz zastępujemy plik docelowy.
                File.Move(temporaryPath, outputPath, overwrite: true);

                ExportProgressBar.Value = 100;
                ExportProgressText.Text = "Eksport zakończony — 100%";
                long outputSize = new FileInfo(outputPath).Length;
                string profileDescription = streamCopy
                    ? "kopiowanie strumieni bez ponownego kodowania"
                    : profile switch
                    {
                        "fast" => "H.264, profil szybki (veryfast, CRF 22), audio AAC 192 kb/s",
                        "small" => "H.264, profil mniejszego pliku (slow, CRF 27), audio AAC 192 kb/s",
                        "quality" => "H.264, profil wysokiej jakości (slow, CRF 16), audio AAC 192 kb/s",
                        _ => "H.264, profil standardowy (medium, CRF 20), audio AAC 192 kb/s"
                    };
                _lastExportSummary = $"Ostatni eksport: {Path.GetFileName(outputPath)} — {profileDescription}, rzeczywisty rozmiar {FormatFileSize(outputSize)}.";
                ExportOperationText.Text = $"Zakończono: {Path.GetFileName(outputPath)} | Rozmiar: {FormatFileSize(outputSize)} | Czas operacji: {_exportStopwatch?.Elapsed.ToString(@"hh\:mm\:ss") ?? "—"}";
                UpdateEstimatedSizeText();
                StatusText.Text = "Zapisano fragment: " + outputPath;

                MessageBox.Show(
                    this,
                    "Fragment został zapisany:\n" + outputPath,
                    "Eksport zakończony",
                    MessageBoxButton.OK,
                    MessageBoxImage.Information);
            }
            catch (OperationCanceledException)
            {
                ExportProgressText.Text = "Eksport anulowany — czyszczenie pliku tymczasowego…";
                ExportOperationText.Text = $"Anulowano operację dla pliku: {_activeOperationName}";
                StatusText.Text =
                    "Eksport anulowano. Plik tymczasowy zostanie usunięty.";
            }
            catch (Exception ex)
            {
                ExportProgressText.Text = "Eksport nie powiódł się.";
                ExportOperationText.Text = $"Błąd podczas przetwarzania: {_activeOperationName}";
                StatusText.Text = "Błąd eksportu.";

                MessageBox.Show(
                    this,
                    ex.Message,
                    "Błąd eksportu",
                    MessageBoxButton.OK,
                    MessageBoxImage.Error);
            }
            finally
            {
                _exportProcess = null;

                process?.Dispose();

                try
                {
                    if (File.Exists(temporaryPath))
                        File.Delete(temporaryPath);
                }
                catch (Exception ex)
                {
                    if (!_closing)
                    {
                        StatusText.Text =
                            "Nie udało się usunąć pliku tymczasowego: " +
                            ex.Message;
                    }
                }

                _exporting = false;
                _exportStopwatch?.Stop();
                _exportStopwatch = null;

                if (!_closing)
                {
                    ExportClipButton.IsEnabled = true;
                    RemoveClipButton.IsEnabled = true;
                    ExportProfileBox.IsEnabled = true;
                    StreamCopyCheckBox.IsEnabled = true;
                    CancelExportButton.IsEnabled = false;
                    CancelExportButton.Visibility = Visibility.Collapsed;
                }
            }
        }

        private async Task ReadFfmpegProgressAsync(
            Process process,
            double duration)
        {
            while (true)
            {
                string? line = await process.StandardOutput.ReadLineAsync();

                if (line == null)
                    break;

                if (line.StartsWith(
                    "out_time_us=",
                    StringComparison.Ordinal))
                {
                    string value = line.Substring("out_time_us=".Length);

                    if (long.TryParse(
                        value,
                        NumberStyles.Integer,
                        CultureInfo.InvariantCulture,
                        out long microseconds))
                    {
                        double seconds = microseconds / 1_000_000.0;

                        double percent = duration > 0
                            ? Math.Clamp(seconds / duration * 100.0, 0, 99.9)
                            : 0;

                        await Dispatcher.InvokeAsync(() =>
                        {
                            if (_closing || _cancelExportRequested)
                                return;

                            ExportProgressBar.Value = percent;

                            double elapsed = _exportStopwatch?.Elapsed.TotalSeconds ?? 0;
                            double speed = elapsed > 0 ? seconds / elapsed : 0;
                            double remaining = speed > 0.001 ? Math.Max(0, duration - seconds) / speed : 0;
                            string eta = double.IsFinite(remaining) && remaining < 864000
                                ? TimeSpan.FromSeconds(remaining).ToString(@"hh\:mm\:ss", CultureInfo.InvariantCulture)
                                : "obliczanie…";
                            ExportProgressText.Text =
                                $"Kodowanie: {percent:0.0}% ({FormatTime(seconds)} / {FormatTime(duration)}) | Pozostało: {eta}";
                        });
                    }
                }
                else if (line == "progress=end")
                {
                    await Dispatcher.InvokeAsync(() =>
                    {
                        if (_closing || _cancelExportRequested)
                            return;

                        ExportProgressBar.Value = 100;
                        ExportProgressText.Text = "Finalizowanie pliku MP4…";
                        ExportOperationText.Text = $"Finalizowanie: {_activeOperationName} | upłynęło: {_exportStopwatch?.Elapsed.ToString(@"hh\:mm\:ss") ?? "—"}";
                    });
                }
            }
        }

        private async void RemoveClip_Click(object sender, RoutedEventArgs e)
        {
            if (_exporting)
            {
                MessageBox.Show("Inna operacja eksportu jest już wykonywana.");
                return;
            }

            if (string.IsNullOrWhiteSpace(_sourceFilePath) ||
                !File.Exists(_sourceFilePath))
            {
                MessageBox.Show("Najpierw otwórz nagranie.");
                return;
            }

            if (!_clipStart.HasValue || !_clipEnd.HasValue)
            {
                MessageBox.Show("Ustaw początek i koniec fragmentu do usunięcia.");
                return;
            }

            double start = _clipStart.Value;
            double end = _clipEnd.Value;

            if (start < 0 || end <= start)
            {
                MessageBox.Show("Nieprawidłowy zakres fragmentu.");
                return;
            }

            string ffmpegPath = _ffmpegService.ExecutablePath;

            if (!File.Exists(ffmpegPath))
            {
                MessageBox.Show("Nie znaleziono pliku:\n" + ffmpegPath);
                return;
            }

            // Odczytaj czas trwania i sprawdź, czy plik ma dźwięk.
            double mediaDuration;
            bool hasAudio;

            try
            {
                var probeInfo = await ProbeMediaAsync(_sourceFilePath);
                mediaDuration = probeInfo.Duration;
                hasAudio = probeInfo.HasAudio;
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Nie udało się odczytać informacji o nagraniu:\n" + ex.Message);
                return;
            }

            const double tolerance = 0.15;

            if (mediaDuration <= 0 ||
                start >= mediaDuration ||
                end > mediaDuration + tolerance)
            {
                MessageBox.Show(
                    $"Zakres wykracza poza długość nagrania ({FormatTime(mediaDuration)}).");
                return;
            }

            end = Math.Min(end, mediaDuration);

            bool keepBefore = start > tolerance;
            bool keepAfter = end < mediaDuration - tolerance;

            if (!keepBefore && !keepAfter)
            {
                MessageBox.Show(
                    "Ten zakres obejmuje całe nagranie. " +
                    "Nie można utworzyć pustego pliku.");
                return;
            }

            string description;

            if (!keepBefore)
                description = $"Usunąć początek do {FormatTime(end)}?";
            else if (!keepAfter)
                description = $"Usunąć koniec od {FormatTime(start)}?";
            else
                description =
                    $"Usunąć fragment od {FormatTime(start)} do {FormatTime(end)}?";

            var confirm = MessageBox.Show(
                description +
                "\n\nPowstanie nowy plik. Oryginał pozostanie bez zmian.",
                "Usuwanie fragmentu",
                MessageBoxButton.YesNo,
                MessageBoxImage.Warning);

            if (confirm != MessageBoxResult.Yes)
                return;

            var dialog = new Microsoft.Win32.SaveFileDialog
            {
                Title = "Zapisz nagranie po usunięciu fragmentu",
                Filter = "Plik MP4 (*.mp4)|*.mp4",
                DefaultExt = ".mp4",
                AddExtension = true,
                FileName = System.IO.Path.GetFileNameWithoutExtension(
                    _sourceFilePath) + "_po_cieciu.mp4"
            };

            if (dialog.ShowDialog() != true)
                return;

            string outputPath = System.IO.Path.GetFullPath(dialog.FileName);
            string sourcePath = System.IO.Path.GetFullPath(_sourceFilePath);

            if (string.Equals(
                outputPath, sourcePath, StringComparison.OrdinalIgnoreCase))
            {
                MessageBox.Show("Plik wynikowy nie może zastępować oryginału.");
                return;
            }

            string temporaryPath = System.IO.Path.Combine(
                System.IO.Path.GetDirectoryName(outputPath)!,
                System.IO.Path.GetFileNameWithoutExtension(outputPath) +
                "_" + Guid.NewGuid().ToString("N") + ".tmp.mp4");

            string F(double value) =>
                value.ToString("0.######", CultureInfo.InvariantCulture);

            // Przycinanie obrazu. W razie potrzeby łączymy dwie pozostałe części.
            string filter;

            if (keepBefore && keepAfter)
            {
                filter =
                    $"[0:v:0]trim=start=0:end={F(start)},setpts=PTS-STARTPTS[v0];" +
                    $"[0:v:0]trim=start={F(end)},setpts=PTS-STARTPTS[v1];" +
                    "[v0][v1]concat=n=2:v=1:a=0[v]";
            }
            else if (keepBefore)
            {
                filter =
                    $"[0:v:0]trim=start=0:end={F(start)}," +
                    "setpts=PTS-STARTPTS[v]";
            }
            else
            {
                filter =
                    $"[0:v:0]trim=start={F(end)},setpts=PTS-STARTPTS[v]";
            }

            // Ścieżkę audio przetwarzamy tylko wtedy, gdy rzeczywiście istnieje.
            if (hasAudio)
            {
                if (keepBefore && keepAfter)
                {
                    filter +=
                        $";[0:a:0]atrim=start=0:end={F(start)},asetpts=PTS-STARTPTS[a0]" +
                        $";[0:a:0]atrim=start={F(end)},asetpts=PTS-STARTPTS[a1]" +
                        ";[a0][a1]concat=n=2:v=0:a=1[a]";
                }
                else if (keepBefore)
                {
                    filter +=
                        $";[0:a:0]atrim=start=0:end={F(start)}," +
                        "asetpts=PTS-STARTPTS[a]";
                }
                else
                {
                    filter +=
                        $";[0:a:0]atrim=start={F(end)}," +
                        "asetpts=PTS-STARTPTS[a]";
                }
            }

            var psi = _ffmpegService.CreateStartInfo(
                redirectStandardOutput: true,
                redirectStandardError: true);

            string[] args =
            {
         "-y",
        "-progress", "pipe:1",
        "-stats_period", "0.5",
        "-nostats",
        "-i", sourcePath,
        "-filter_complex", filter,
        "-map", "[v]"
    };

            if (hasAudio)
            {
                args = args.Concat(new[] { "-map", "[a]" }).ToArray();
            }

            args = args.Concat(new[]
            {
        "-c:v", "libx264",
        "-preset", "fast",
        "-crf", "20"
    }).ToArray();

            if (hasAudio)
            {
                args = args.Concat(new[] { "-c:a", "aac", "-b:a", "192k" }).ToArray();
            }

            args = args.Concat(new[]
            {
        "-movflags", "+faststart",
        temporaryPath
    }).ToArray();

            foreach (string arg in args)
                psi.ArgumentList.Add(arg);

            Process? process = null;

            try
            {
                _exporting = true;
                _cancelExportRequested = false;

                RemoveClipButton.IsEnabled = false;
                ExportClipButton.IsEnabled = false;

                ExportProgressBar.Value = 0;
                ExportProgressText.Text = "Przygotowywanie nagrania…";
                ExportProgressBar.Visibility = Visibility.Visible;
                ExportProgressText.Visibility = Visibility.Visible;
                CancelExportButton.Visibility = Visibility.Visible;
                CancelExportButton.IsEnabled = true;

                StatusText.Text = "Usuwanie fragmentu nagrania…";

                process = new Process
                {
                    StartInfo = psi,
                    EnableRaisingEvents = true
                };

                process.Start();
                _exportProcess = process;

                // Czytamy oba strumienie równolegle, aby nie blokować FFmpeg.
                Task progressTask = ReadFfmpegProgressAsync(
                    process,
                    mediaDuration - (end - start));

                Task<string> errorTask =
                    process.StandardError.ReadToEndAsync();

                await process.WaitForExitAsync();
                await progressTask;

                string errors = await errorTask;

                if (_cancelExportRequested)
                    throw new OperationCanceledException("Eksport anulowano.");

                if (process.ExitCode != 0)
                    throw new Exception("FFmpeg zgłosił błąd:\n" + errors);

                if (!File.Exists(temporaryPath) ||
                    new FileInfo(temporaryPath).Length == 0)
                {
                    throw new Exception("Nie utworzono prawidłowego pliku wynikowego.");
                }

                File.Move(temporaryPath, outputPath, overwrite: true);

                StatusText.Text = "Zapisano nagranie po usunięciu fragmentu.";

                MessageBox.Show(
                    $"Operacja zakończona.\n\nPlik:\n{outputPath}\n\n" +
                    (hasAudio ? "Dźwięk został zachowany." :
                        "Nagranie nie zawierało dźwięku — zapisano sam obraz.") +
                    "\nOryginał nie został zmieniony.",
                    "Gotowe",
                    MessageBoxButton.OK,
                    MessageBoxImage.Information);
            }

            catch (OperationCanceledException)
            {
                StatusText.Text = "Operacja anulowana.";
                ExportProgressText.Text = "Anulowano usuwanie fragmentu.";
            }

            catch (Exception ex)
            {
                StatusText.Text = "Błąd usuwania fragmentu.";

                MessageBox.Show(
                    ex.Message,
                    "Błąd",
                    MessageBoxButton.OK,
                    MessageBoxImage.Error);
            }
            finally
            {
                process?.Dispose();

                try
                {
                    if (File.Exists(temporaryPath))
                        File.Delete(temporaryPath);
                }
                catch (IOException)
                {
                    // Jeśli plik jest zablokowany, pozostaje do ręcznego usunięcia.
                }

                _exportProcess = null;

                ExportProgressBar.Value = 0;
                ExportProgressBar.Visibility = Visibility.Collapsed;
                ExportProgressText.Visibility = Visibility.Collapsed;
                CancelExportButton.Visibility = Visibility.Collapsed;
                CancelExportButton.IsEnabled = false;

                RemoveClipButton.IsEnabled = true;
                ExportClipButton.IsEnabled = true;
                _exporting = false;
            }
        }

                    MessageBoxImage.Warning);

                {
                    e.Cancel = true;

                try
                {
                    if (_exportProcess != null && !_exportProcess.HasExited)
                        _exportProcess.Kill(entireProcessTree: true);

                catch (InvalidOperationException) { }
                catch (System.ComponentModel.Win32Exception) { }
            }

            _closing = true;
            try
            {
                if (_exportProcess != null && !_exportProcess.HasExited)
                    _exportProcess.Kill(entireProcessTree: true);
            }
            catch (InvalidOperationException)
            {
            }
            catch (System.ComponentModel.Win32Exception)
            {
            }

            _ready = false;
            _ = _mpvService.DisposeAsync();
        }

        private void UpdateEstimatedSizeText()
        {
            if (_currentMediaInfo == null || string.IsNullOrWhiteSpace(_sourceFilePath)) return;
            double fullDuration = _currentMediaInfo.Duration;
            bool hasRange = _clipStart.HasValue && _clipEnd.HasValue && _clipEnd.Value > _clipStart.Value;
            double selectedDuration = hasRange ? _clipEnd!.Value - _clipStart!.Value : fullDuration;
            string estimate = fullDuration > 0
                ? FormatFileSize((long)(_currentMediaInfo.FileSize * Math.Clamp(selectedDuration / fullDuration, 0, 1)))
                : "nie można oszacować";
            string rangeLabel = hasRange ? $"zaznaczenie ({FormatTime(selectedDuration)})" : "całe nagranie";
            MediaInfoText.Text =
                $"Źródło: {_currentMediaInfo.VideoCodec}, {_currentMediaInfo.Resolution}, {_currentMediaInfo.Fps} FPS, bitrate {_currentMediaInfo.Bitrate}, audio: {_currentMediaInfo.AudioCodec}; " +
                $"czas {FormatTime(fullDuration)}, rozmiar {FormatFileSize(_currentMediaInfo.FileSize)}.\n" +
                $"Szacowany rozmiar {rangeLabel} przy podobnym bitrate: około {estimate}. To przybliżenie; rozmiar MP4 po ponownym kodowaniu może się różnić." +
                (string.IsNullOrWhiteSpace(_lastExportSummary) ? "" : "\n" + _lastExportSummary);
        }

        private static string FormatFileSize(long bytes)
        {
            if (bytes < 0) bytes = 0;
            string[] units = { "B", "KB", "MB", "GB", "TB" };
            double size = bytes;
            int unit = 0;
            while (size >= 1024 && unit < units.Length - 1) { size /= 1024; unit++; }
            return $"{size:0.##} {units[unit]}";
        }

        private void HandleMessage(JsonElement message)
        {
            if (_closing)
                return;

            // Odpowiedź na nasze polecenie zapisu klatki.
            if (message.TryGetProperty("request_id", out JsonElement requestId) &&
                requestId.ValueKind == JsonValueKind.Number &&
                requestId.TryGetInt32(out int id) &&
                id == ScreenshotRequestId)
            {
                string? savedPath = _pendingScreenshotPath;

                _pendingScreenshotPath = null;
                ScreenshotButton.IsEnabled = true;

                bool success =
                    message.TryGetProperty("error", out JsonElement result) &&
                    result.ValueKind == JsonValueKind.String &&
                    result.GetString() == "success";

                if (success)
                {
                    if (!string.IsNullOrWhiteSpace(savedPath))
                        AddRecentScreenshot(savedPath);
                    StatusText.Text = "Zapisano klatkę: " + savedPath;
                }
                else
                {
                    string errorText =
                        message.TryGetProperty("error", out JsonElement errorValue)
                            ? errorValue.ToString()
                            : "Brak informacji o błędzie.";

                    StatusText.Text = "Nie udało się zapisać klatki.";

                    MessageBox.Show(
                        this,
                        "Silnik mpv nie zapisał obrazu:\n" + errorText,
                        "Błąd zapisu klatki",
                        MessageBoxButton.OK,
                        MessageBoxImage.Error);
                }

                return;
            }

            // Odpowiedź na polecenie.
            if (!message.TryGetProperty("event", out JsonElement ev))
            {
                if (message.TryGetProperty(
                        "error", out JsonElement error) &&
                    error.ValueKind == JsonValueKind.String &&
                    error.GetString() != "success")
                {
                    StatusText.Text =
                        "mpv: " + error.GetString();
                }

                return;
            }

            if (ev.GetString() != "property-change")
                return;

            if (!message.TryGetProperty("name", out JsonElement name) ||
                !message.TryGetProperty("data", out JsonElement data))
            {
                return;
            }

            switch (name.GetString())
            {
                case "duration":
                    if (data.ValueKind == JsonValueKind.Number &&
                        data.TryGetDouble(out double duration) &&
                        double.IsFinite(duration) &&
                        duration > 0)
                    {
                        SetDuration(duration);
                    }
                    break;

                case "time-pos":
                    if (data.ValueKind == JsonValueKind.Number &&
                        data.TryGetDouble(out double position) &&
                        double.IsFinite(position) &&
                        !_dragging)
                    {
                        _updatingSlider = true;

                        try
                        {
                            PositionSlider.Value = Math.Clamp(
                                position,
                                PositionSlider.Minimum,
                                PositionSlider.Maximum);
                        }
                        finally
                        {
                            _updatingSlider = false;
                        }

                        PositionText.Text = FormatTime(position);
                    }
                    break;

                case "eof-reached":
                    _eofReached =
                        data.ValueKind == JsonValueKind.True;

                    StatusText.Text = _eofReached
                        ? "Koniec filmu. Przewiń suwakiem albo naciśnij Play."
                        : _fileName;
                    break;

                case "pause":
                    if (data.ValueKind == JsonValueKind.True)
                        PlayButton.Content = "Play";
                    else if (data.ValueKind == JsonValueKind.False)
                        PlayButton.Content = "Pauza";
                    break;
            }
        }

        private void SetDuration(double duration)
        {
            _updatingSlider = true;

            try
            {
                PositionSlider.Maximum = duration;
                PositionSlider.IsEnabled = true;
            }
            finally
            {
                _updatingSlider = false;
            }

            DurationText.Text = FormatTime(duration);
            UpdateTimelineOverlay();
            UpdateTimelineOverlay();
        }

        private static string FormatTime(double seconds)
        {
            TimeSpan time = TimeSpan.FromSeconds(
                Math.Max(0, seconds));

            return string.Format(
                CultureInfo.InvariantCulture,
                "{0:00}:{1:00}:{2:00}",
                (int)time.TotalHours,
                time.Minutes,
                time.Seconds);
        }

    object sender,
    RoutedEventArgs e)
        {

            if (!_clipStart.HasValue)

        private static string FormatClipTime(double seconds)
        {
            return TimeSpan
                .FromSeconds(Math.Max(0, seconds))
                .ToString(
                    @"d\.hh\:mm\:ss\.fff",
                    CultureInfo.InvariantCulture);
        }

        private void UpdateClipRangeText()
        {
            string start = _clipStart.HasValue
                ? FormatClipTime(_clipStart.Value)
                : "—";

            string end = _clipEnd.HasValue
                ? FormatClipTime(_clipEnd.Value)
                : "—";

            string duration = "—";

            if (_clipStart.HasValue &&
                _clipEnd.HasValue &&
                _clipEnd.Value > _clipStart.Value)
            {
                duration = FormatClipTime(_clipEnd.Value - _clipStart.Value);
            }

            ClipRangeText.Text =
                $"Początek: {start}\n" +
                $"Koniec: {end}\n" +
                $"Długość: {duration}";

            UpdateTimelineOverlay();
            UpdateEstimatedSizeText();
        }

        private void TimelineOverlay_SizeChanged(
            object sender,
            SizeChangedEventArgs e)
        {
            UpdateTimelineOverlay();
        }

        private void UpdateTimelineOverlay()
        {
            if (TimelineOverlay == null ||
                ClipStartHandle == null ||
                ClipEndHandle == null ||
                ClipSelectionRect == null)
            {
                return;
            }

            double width = TimelineOverlay.ActualWidth;
            double duration = PositionSlider.Maximum - PositionSlider.Minimum;

            if (width <= 0 || duration <= 0 || !double.IsFinite(width))
                return;

            bool hasStart = _clipStart.HasValue;
            bool hasEnd = _clipEnd.HasValue;
            bool hasRange = hasStart && hasEnd && _clipEnd!.Value > _clipStart!.Value;

            ClipStartHandle.Visibility = hasStart ? Visibility.Visible : Visibility.Collapsed;
            ClipEndHandle.Visibility = hasEnd ? Visibility.Visible : Visibility.Collapsed;
            ClipSelectionRect.Visibility = hasRange ? Visibility.Visible : Visibility.Collapsed;

            double startX = hasStart
                ? Math.Clamp((_clipStart!.Value - PositionSlider.Minimum) / duration * width, 0, width)
                : 0;
            double endX = hasEnd
                ? Math.Clamp((_clipEnd!.Value - PositionSlider.Minimum) / duration * width, 0, width)
                : 0;

            if (hasStart)
                Canvas.SetLeft(ClipStartHandle, Math.Clamp(startX - ClipStartHandle.Width / 2, -1, Math.Max(-1, width - ClipStartHandle.Width + 1)));

            if (hasEnd)
                Canvas.SetLeft(ClipEndHandle, Math.Clamp(endX - ClipEndHandle.Width / 2, -1, Math.Max(-1, width - ClipEndHandle.Width + 1)));

            if (hasRange)
            {
                Canvas.SetLeft(ClipSelectionRect, startX);
                ClipSelectionRect.Width = Math.Max(0, endX - startX);
                Canvas.SetTop(ClipSelectionRect, (TimelineOverlay.ActualHeight - ClipSelectionRect.Height) / 2);
            }

            DrawTimelineMarkers();
        }

        private void ClipStartHandle_DragDelta(
            object sender,
            DragDeltaEventArgs e)
        {
            if (_clipStart == null || TimelineOverlay.ActualWidth <= 0)
                return;

            double width = TimelineOverlay.ActualWidth;
            double left = Canvas.GetLeft(ClipStartHandle);
            if (!double.IsFinite(left)) left = 0;

            double center = Math.Clamp(
                left + ClipStartHandle.Width / 2 + e.HorizontalChange,
                0,
                width);

            double newTime = PositionSlider.Minimum +
                center / width * (PositionSlider.Maximum - PositionSlider.Minimum);

            if (_clipEnd.HasValue)
                newTime = Math.Min(newTime, _clipEnd.Value - 0.001);

            _clipStart = Math.Clamp(newTime, PositionSlider.Minimum, PositionSlider.Maximum);
            UpdateClipRangeText();
        }

        private void ClipEndHandle_DragDelta(
            object sender,
            DragDeltaEventArgs e)
        {
            if (_clipEnd == null || TimelineOverlay.ActualWidth <= 0)
                return;

            double width = TimelineOverlay.ActualWidth;
            double left = Canvas.GetLeft(ClipEndHandle);
            if (!double.IsFinite(left)) left = 0;

            double center = Math.Clamp(
                left + ClipEndHandle.Width / 2 + e.HorizontalChange,
                0,
                width);

            double newTime = PositionSlider.Minimum +
                center / width * (PositionSlider.Maximum - PositionSlider.Minimum);

            if (_clipStart.HasValue)
                newTime = Math.Max(newTime, _clipStart.Value + 0.001);

            _clipEnd = Math.Clamp(newTime, PositionSlider.Minimum, PositionSlider.Maximum);
            UpdateClipRangeText();
        }

        private void DrawTimelineMarkers()

using System;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.Text.RegularExpressions;
using System.Threading.Tasks;

namespace VideoPlayer
{
    internal sealed class MediaInfoService
    {
        private readonly string _ffmpegPath;
        public MediaInfoService(string ffmpegPath) => _ffmpegPath = ffmpegPath;

        public async Task<MediaInfoData> ReadAsync(string sourcePath)
        {
            if (!File.Exists(_ffmpegPath))
                throw new FileNotFoundException("Brak ffmpeg.exe — nie można odczytać parametrów nagrania.", _ffmpegPath);

            var psi = new ProcessStartInfo
            {
                FileName = _ffmpegPath,
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardError = true
            };
            psi.ArgumentList.Add("-hide_banner");
            psi.ArgumentList.Add("-i");
            psi.ArgumentList.Add(sourcePath);

            using var process = new Process { StartInfo = psi };
            process.Start();
            string output = await process.StandardError.ReadToEndAsync().ConfigureAwait(false);
            await process.WaitForExitAsync().ConfigureAwait(false);

            var info = new MediaInfoData
            {
                FileSize = new FileInfo(sourcePath).Length,
                Duration = ParseDuration(output)
            };
            var videoLine = Regex.Match(output, @"Stream #\S+.*?: Video:\s*([^, ]+)(.*)");
            if (videoLine.Success)
            {
                info.VideoCodec = videoLine.Groups[1].Value;
                string details = videoLine.Groups[2].Value;
                var resolution = Regex.Match(details, @"(\d{2,5})x(\d{2,5})");
                if (resolution.Success) info.Resolution = resolution.Groups[1].Value + "×" + resolution.Groups[2].Value;
                var fps = Regex.Match(details, @"([\d.]+)\s*fps");
                if (fps.Success) info.Fps = fps.Groups[1].Value;
            }
            var audio = Regex.Match(output, @"Stream #\S+.*?: Audio:\s*([^, ]+)");
            if (audio.Success) info.AudioCodec = audio.Groups[1].Value;
            var bitrate = Regex.Match(output, @"bitrate:\s*(\d+)\s*kb/s");
            if (bitrate.Success) info.Bitrate = bitrate.Groups[1].Value + " kb/s";
            return info;
        }

        public static double ParseDuration(string output)
        {
            var match = Regex.Match(output, @"Duration:\s*(\d+):(\d+):(\d+(?:\.\d+)?)");
            if (!match.Success) return 0;
            return double.Parse(match.Groups[1].Value, CultureInfo.InvariantCulture) * 3600 +
                   double.Parse(match.Groups[2].Value, CultureInfo.InvariantCulture) * 60 +
                   double.Parse(match.Groups[3].Value, CultureInfo.InvariantCulture);
        }
    }

    internal sealed class MediaInfoData
    {
        public double Duration { get; set; }
        public string VideoCodec { get; set; } = "nieznany";
        public string Resolution { get; set; } = "nieznana";
        public string Fps { get; set; } = "nieznane";
        public string Bitrate { get; set; } = "nieznany";
        public string AudioCodec { get; set; } = "brak / nieznany";
        public long FileSize { get; set; }
    }
}

using System;
using System.Diagnostics;
using System.IO;
using System.IO.Pipes;
using System.Text;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;

namespace VideoPlayer
{
    /// <summary>Obsługuje proces mpv i jego interfejs IPC.</summary>
    internal sealed class MpvService : IAsyncDisposable
    {
        private readonly string _pipeName = "VideoPlayer_" + Guid.NewGuid().ToString("N");
        private readonly SemaphoreSlim _sendLock = new(1, 1);
        private Process? _process;
        private NamedPipeClientStream? _pipe;
        private StreamWriter? _writer;
        private bool _disposed;

        public bool IsReady { get; private set; }
        public event Action<JsonElement>? MessageReceived;

        public async Task StartAsync(string executable, IntPtr windowHandle)
        {
            if (!File.Exists(executable))
                throw new FileNotFoundException("Nie znaleziono mpv.exe w folderze mpv.", executable);

            var startInfo = new ProcessStartInfo
            {
                FileName = executable,
                UseShellExecute = false,
                CreateNoWindow = true
            };

            string[] arguments =
            {
                "--no-config", "--idle=yes", "--keep-open=always",
                "--keep-open-pause=yes", "--terminal=no", "--osc=no",
                "--input-default-bindings=no", "--input-vo-keyboard=no",
                "--wid=" + unchecked((uint)windowHandle.ToInt64()).ToString(System.Globalization.CultureInfo.InvariantCulture),
                "--screenshot-jpeg-quality=95",
                "--input-ipc-server=\\\\.\\pipe\\" + _pipeName
            };
            foreach (string argument in arguments)
                startInfo.ArgumentList.Add(argument);

            _process = Process.Start(startInfo)
                ?? throw new InvalidOperationException("Nie udało się uruchomić mpv.");

            _pipe = new NamedPipeClientStream(".", _pipeName, PipeDirection.InOut, PipeOptions.Asynchronous);
            await _pipe.ConnectAsync(10000).ConfigureAwait(false);
            _writer = new StreamWriter(_pipe, new UTF8Encoding(false), 4096, leaveOpen: true)
            {
                AutoFlush = true,
                NewLine = "\n"
            };
            IsReady = true;
            _ = ReadMessagesAsync(_pipe);
        }

        public async Task SendAsync(params object[] command)
        {
            await SendMessageAsync(new { command }).ConfigureAwait(false);
        }

        public async Task SendRequestAsync(int requestId, params object[] command)
        {
            await SendMessageAsync(new { command, request_id = requestId }).ConfigureAwait(false);
        }

        private async Task SendMessageAsync(object payload)
        {
            if (!IsReady || _disposed) return;
            await _sendLock.WaitAsync().ConfigureAwait(false);
            try
            {
                if (_writer == null || !IsReady || _disposed) return;
                await _writer.WriteLineAsync(JsonSerializer.Serialize(payload)).ConfigureAwait(false);
            }
            catch (Exception ex) when (ex is IOException || ex is ObjectDisposedException)
            {
                IsReady = false;
                throw;
            }
            finally
            {
                _sendLock.Release();
            }
        }

        private async Task ReadMessagesAsync(NamedPipeClientStream pipe)
        {
            try
            {
                using var reader = new StreamReader(pipe, Encoding.UTF8, false, 4096, leaveOpen: true);
                while (!_disposed)
                {
                    string? line = await reader.ReadLineAsync().ConfigureAwait(false);
                    if (line == null) break;
                    try
                    {
                        using JsonDocument document = JsonDocument.Parse(line);
                        MessageReceived?.Invoke(document.RootElement.Clone());
                    }
                    catch (JsonException) { }
                }
            }
            catch (Exception ex) when (ex is IOException || ex is ObjectDisposedException) { }
            finally { IsReady = false; }
        }

        public async ValueTask DisposeAsync()
        {
            if (_disposed) return;
            _disposed = true;
            IsReady = false;
            try { _pipe?.Dispose(); } catch { }
            try
            {
                if (_process != null && !_process.HasExited)
                    _process.Kill(entireProcessTree: true);
            }
            catch (InvalidOperationException) { }
            catch (System.ComponentModel.Win32Exception) { }
            _process?.Dispose();
            _writer?.Dispose();
            _sendLock.Dispose();
            await Task.CompletedTask;
        }
    }
}

# VideoPlayer — Etap 6 i Etap 7

## Zmiany

### Etap 6 — usługi
- `MpvService.cs`: uruchamianie procesu mpv, IPC przez named pipe, wysyłanie poleceń i przekazywanie komunikatów.
- `FfmpegService.cs`: wspólna ścieżka do FFmpeg, konfiguracja procesów oraz odczyt czasu trwania i obecności audio.
- `MediaInfoService.cs`: odczyt kodeków, rozdzielczości, FPS, bitrate'u, czasu i rozmiaru nagrania.
- `HistoryService.cs`: historia ostatnich nagrań/zrzutów oraz trwały zapis znaczników zdarzeń.

`MainWindow` nadal koordynuje interfejs, a eksport zachowuje dotychczasową logikę postępu, anulowania i profili. Wspólne tworzenie procesów FFmpeg i wykrywanie właściwości mediów korzysta już z usług.

### Etap 7 — znaczniki zdarzeń
- dodawanie znacznika w aktualnej pozycji odtwarzania;
- wymagany opis zdarzenia;
- zmiana opisu, usuwanie i przejście do wybranego znacznika;
- dwuklik na pozycji listy lub kliknięcie pomarańczowego punktu na osi czasu przechodzi do czasu zdarzenia;
- znaczniki są zapisywane osobno dla każdego pliku w `%LOCALAPPDATA%\VideoPlayer\timeline-markers.json`.

Historia nagrań i zrzutów pozostaje w `%LOCALAPPDATA%\VideoPlayer\history.json`, w zgodnym formacie JSON.

## Wdrożenie
1. Rozpakuj źródła i otwórz `VideoPlayer.slnx` albo `VideoPlayer.csproj` w Visual Studio.
2. Upewnij się, że w katalogu projektu znajduje się `mpv\mpv.exe`.
3. Upewnij się, że w katalogu projektu znajduje się `ffmpeg\ffmpeg.exe` (oraz pozostałe pliki FFmpeg, jeśli używasz ich w swoim wdrożeniu).
4. Skompiluj projekt dla `net8.0-windows`.
5. Przetestuj otwarcie nagrania, pauzę/przewijanie, zapis klatki, eksport i anulowanie, a następnie dodawanie/edycję/usuwanie znaczników i ponowne otwarcie tego samego nagrania.

## Weryfikacja w tym środowisku
- Plik XAML przeszedł kontrolę poprawności XML.
- Sprawdzono, że nazwy obsługiwanych zdarzeń z XAML mają odpowiadające metody w kodzie.
- Sprawdzono bilans nawiasów klamrowych w plikach C#.
- **Nie wykonano kompilacji WPF**: w środowisku roboczym nie ma `dotnet`/MSBuild ani narzędzi Windows WPF. Dlatego proszę wykonać kompilację i test funkcjonalny w Visual Studio przed użyciem produkcyjnym.

## Dodatkowa funkcja: regulacja obszaru obrazu i skalowania

- Wysokość obszaru wideo można zmieniać przez przeciąganie poziomego separatora między obrazem a panelem sterowania.
- Lista „Skalowanie” w prawym górnym rogu podglądu udostępnia tryby: „Dopasuj cały obraz”, „Wypełnij okno”, 125%, 150% i 200%.
- „Dopasuj cały obraz” pokazuje pełną klatkę bez celowego przycinania; „Wypełnij okno” wypełnia obszar i może przyciąć krawędzie. Powiększenia 125–200% mogą również obciąć fragment obrazu poza obszarem podglądu.
- Rozmiar obszaru zmienia tylko układ aplikacji, a skalowanie jest ustawiane w mpv. Oryginalny plik wideo nie jest modyfikowany.

## Przenośne dane SQLite

Historia ostatnio otwieranych nagrań, historia zrzutów ekranu oraz znaczniki osi czasu są teraz przechowywane w pliku `VideoPlayer.db` w katalogu aplikacji (obok pliku EXE). Nie jest już potrzebny katalog `%LOCALAPPDATA%\VideoPlayer` do bieżącego zapisu tych danych.

- Aby przenieść aplikację, skopiuj cały katalog aplikacji razem z `VideoPlayer.db`, folderami `mpv` i `ffmpeg`.
- Uruchamiaj aplikację z katalogu, w którym konto użytkownika ma prawo zapisu. Katalog `C:\Program Files` może być chroniony; dla wersji przenośnej użyj np. własnego katalogu lub pamięci USB.
- Przy pierwszym uruchomieniu nowej wersji aplikacja próbuje jednorazowo zaimportować dotychczasowe `history.json` i `timeline-markers.json` z `%LOCALAPPDATA%\VideoPlayer`. Stare pliki JSON nie są usuwane.
- Historia przechowuje ścieżki do nagrań i zrzutów, a nie same pliki multimedialne. Po przeniesieniu komputera lub zmianie liter dysków pozycje historii mogą wymagać ponownego otwarcia plików z ich nowej lokalizacji.
